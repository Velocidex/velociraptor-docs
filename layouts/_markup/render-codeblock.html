{{- /* Code fence render hook (Hextra chrome + copy button). VQL fences are pre-rendered at build time by scripts/vql_highlight (which uses a Chroma lexer whose output matches Hugo's own highlight function) and looked up here by sha256 of the fence content. The generated/vql directory is generated content (gitignored) — run `make vql_highlight` after editing marked-up VQL. If a fence has no pre-rendered file the vql/VQL language is aliased to Chroma's SQL lexer (VQL is SQL-derived), so pages still build with correct syntax highlighting; the data-vql="fallback" attribute lets CI detect a stale generation. Everything else falls through to Hextra's codeblock component. */ -}} {{- $lang := .Attributes.lang | default .Type -}} {{- $isVQL := or (eq $lang "vql") (eq $lang "VQL") -}} {{- $prerendered := "" -}} {{- if $isVQL -}} {{- $file := printf "generated/vql/%s.html" (sha256 .Inner) -}} {{- with os.Stat $file -}} {{- $prerendered = readFile $file -}} {{- end -}} {{- end -}} {{- if $prerendered -}}
{{- with .Attributes.filename -}}
{{- with $.Attributes.base_url -}} {{- $file_url := urls.JoinPath (strings.TrimSuffix "/" .) (strings.TrimPrefix "/" $) -}} {{- $ -}} {{- partial "utils/icon" (dict "name" "external-link" "attributes" "height=1em") -}} {{- else -}} {{- . -}} {{- end -}}
{{- end -}}
{{- $prerendered | safeHTML -}}
{{- if or (eq site.Params.highlight.copy.enable nil) (site.Params.highlight.copy.enable) -}} {{- partialCached "components/codeblock-copy-button" (dict "filename" (.Attributes.filename | default "")) (.Attributes.filename | default "") -}} {{- end -}}
{{- else -}} {{- if $isVQL -}} {{- $lang = "sql" -}} {{- end -}} {{- $filename := .Attributes.filename | default "" -}} {{- $baseURL := .Attributes.base_url | default "" -}}
{{- partial "components/codeblock" (dict "filename" $filename "lang" $lang "base_url" $baseURL "content" .Inner "options" .Options) -}} {{- if or (eq site.Params.highlight.copy.enable nil) (site.Params.highlight.copy.enable) -}} {{- partialCached "components/codeblock-copy-button" (dict "filename" $filename) $filename -}} {{- end -}}
{{- end -}}
